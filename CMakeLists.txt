# Snapshot-vault change: this copy omits assets/ (original game data) and reference/
# (recorded x86 comparison dumps). Contract tests replay those trees, so a test is only
# registered when its command does not need them, or when both are present at the source
# root. See README.md for how to link the excluded trees back in.
set(FSB_ORIGINAL_FIXTURES OFF)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/reference")
    set(FSB_ORIGINAL_FIXTURES ON)
endif()
function(add_test)
    if(NOT FSB_ORIGINAL_FIXTURES)
        foreach(arg IN LISTS ARGV)
            if(arg MATCHES "/(assets|reference)(/|$)")
                return()
            endif()
        endforeach()
    endif()
    _add_test(${ARGV})
    set_property(GLOBAL APPEND PROPERTY FSB_REGISTERED_TESTS "${ARGV1}")
endfunction()

# Skipped tests have no properties to set; drop their names instead of failing configure.
function(set_tests_properties)
    get_property(registered GLOBAL PROPERTY FSB_REGISTERED_TESTS)
    set(kept "")
    foreach(arg IN LISTS ARGV)
        if(arg STREQUAL "PROPERTIES")
            break()
        endif()
        if(arg IN_LIST registered)
            list(APPEND kept "${arg}")
        endif()
    endforeach()
    if(kept)
        list(FIND ARGV "PROPERTIES" properties_index)
        list(SUBLIST ARGV ${properties_index} -1 tail)
        _set_tests_properties(${kept} ${tail})
    endif()
endfunction()

    # needs several of those, which one --preload-file cannot produce.

        set(FSB_WEB_PRELOAD --preload-file "${FSB_WEB_ASSETS}@/assets")

        -sFORCE_FILESYSTEM=1 ${FSB_WEB_PRELOAD})
    set_target_properties(fsb_event0_web PROPERTIES SUFFIX ".js")
    # POST_BUILD would only run when the target relinks, which silently serves a
    # stale page after a JS-only edit. Depending on the files themselves does not.
    set(FSB_WEB_PAGE_FILES "")
    foreach(page index.html audio-worklet.js)
        add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${page}"
            COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/tools/web/${page}" "${CMAKE_CURRENT_BINARY_DIR}/${page}"
            DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tools/web/${page}")
        list(APPEND FSB_WEB_PAGE_FILES "${CMAKE_CURRENT_BINARY_DIR}/${page}")
    endforeach()
    add_custom_target(fsb_event0_web_page ALL DEPENDS ${FSB_WEB_PAGE_FILES})
    add_dependencies(fsb_event0_web fsb_event0_web_page)
